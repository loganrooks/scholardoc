from scholargt.schema import (
    DocumentGT, DocumentSource, DocumentStructure, DocumentRelationships,
    Footnote, Citation, Section, MarkerInfo, ContentSpan,
    SemanticElement, SCHEMA_VERSION
)
from scholargt.schema.labels import CitationType

doc = DocumentGT(
    document_id='test_doc',
    source=DocumentSource(pdf='test.pdf', title='Being and Time', author='Martin Heidegger'),
    elements=[
        Footnote(id='fn1', marker=MarkerInfo(text='1', page=0),
                 content=[ContentSpan(page=0, text='See analysis'), ContentSpan(page=1, text='continued', is_continuation=True)],
                 location='page_bottom'),
        Citation(id='ct1', raw_text='(SZ, 41)', citation_type=CitationType.ABBREVIATED),
    ],
    structure=DocumentStructure(sections=['sec1']),
)
j = doc.model_dump_json(indent=2)
doc2 = DocumentGT.model_validate_json(j)
assert len(doc2.elements) == 2
assert isinstance(doc2.elements[0], Footnote)
print(f'DocumentGT: {doc2.document_id}, {len(doc2.elements)} elements, version={doc2.schema_version}')
print('Round-trip OK')
