  ```python
  class Note(GTElement):
      """A scholarly note — footnote, endnote, or marginal note.

      Replaces separate Footnote and Endnote models. TEI P5 uses the same
      approach: one <note> element with @place distinguishing location.
      Differences between footnotes and endnotes are properties (placement,
      scope), not separate types.
      """
      element_type: Literal["note"] = "note"
      marker_text: str | None = Field(
          default=None, description="Marker text as displayed (e.g., '1', '*', 'a')"
      )
      body_marker: LocationRef = Field(
          description="Where the reference marker appears in body text"
      )
      content_marker: LocationRef | None = Field(
          default=None,
          description="Where note content begins (critical for endnotes)"
      )
      content: list[ContentSpan] = Field(
          description="Content text spans (may cross pages)"
      )
      placement: Literal["page_bottom", "end_of_chapter", "end_of_book", "margin"] = Field(
          description="Where the note content appears"
      )
      scope: Literal["page", "chapter", "section", "essay", "document"] | None = Field(
          default=None, description="Numbering reset boundary"
      )
      note_source: Literal["author", "translator", "editor"] | None = Field(
          default=None, description="Who wrote the note"
      )
      note_schema_id: str | None = Field(
          default=None, description="Reference to NoteSchema ID (if document defines schemas)"
      )
  ```

  **NEW element model — Commentary:**
  ```python
  class Commentary(GTElement):
      """Commentary organized by passage reference, not by markers.

      Separate from Note: commentary is organized by passage (using canonical
      coordinates like Stephanus pages), not by marker positions. Supports
      multi-layer commentary (e.g., Talmud-style) where multiple commentaries
      reference the same passage.

      Informed by TEI critical apparatus (app/lem/rdg) and location-referenced method.
      """
      element_type: Literal["commentary"] = "commentary"
      source: str = Field(description="Who wrote the commentary (e.g., 'Rashi', 'editor')")
      passage_ref: str = Field(description="What passage it comments on (canonical coordinates)")
      reference_system: ReferenceSystem = Field(description="How it locates the passage")
      target_location: LocationRef | None = Field(
          default=None, description="Where in GT the target passage is"
      )
      content: list[ContentSpan] = Field(description="Commentary content spans")
      layer: str | None = Field(
          default=None, description="Commentary layer identifier (for multi-layer commentary)"
      )
  ```
