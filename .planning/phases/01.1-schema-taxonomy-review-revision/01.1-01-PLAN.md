  ```python
  class CitationFormat(str, Enum):
      """How a citation appears visually in text."""
      PARENTHETICAL = "parenthetical"     # (Smith, 2020)
      NUMERIC = "numeric"                  # [1] or superscript number
      INLINE_AUTHOR = "inline_author"      # Smith argues...
      NOTE_BASED = "note_based"            # Chicago NB: footnote IS the citation
      AUTHOR_TITLE = "author_title"        # Author, Short Title

  class ReferenceSystem(str, Enum):
      """Canonical reference system used by citation or marginal reference.

      Named values document well-established systems the schema handles.
      Use CUSTOM for project-specific systems not listed here.
      """
      STANDARD = "standard"               # Author + Year + Page
      STEPHANUS = "stephanus"              # Plato: dialogue + page + section (e.g., Republic 514a)
      BEKKER = "bekker"                    # Aristotle: page + column + line (e.g., Met. 1003a21)
      AKADEMIE = "akademie"               # Kant: work + volume + page (e.g., KpV Ak V:30)
      AB_EDITION = "ab_edition"            # Kant KrV: A/B edition pages (e.g., A51/B75)
      PARAGRAPH = "paragraph"              # Hegel: section number + R + A (e.g., PhR §135R)
      SZ_PAGINATION = "sz_pagination"      # Heidegger: SZ page numbers
      DIELS_KRANZ = "diels_kranz"          # Pre-Socratics: philosopher# + A/B/C + fragment#
      LINE_NUMBER = "line_number"          # Poetry, classical texts: work + line/verse
      CHAPTER_VERSE = "chapter_verse"      # Biblical: Book + chapter:verse
      LEGAL = "legal"                      # Legal: § paragraph, ¶, Bluebook pincites
      CUSTOM = "custom"                    # Project-specific system

  class CitationStyle(str, Enum):
      """Document-level citation convention — the third axis of citation decomposition.

      Completes the orthogonal decomposition: CitationFormat (per-citation visual)
      × ReferenceSystem (per-citation coordinates) × CitationStyle (document-level convention).
      """
      APA = "apa"                          # American Psychological Association
      CHICAGO_NB = "chicago_nb"            # Chicago Notes-Bibliography
      CHICAGO_AD = "chicago_ad"            # Chicago Author-Date
      MLA = "mla"                          # Modern Language Association
      VANCOUVER = "vancouver"              # Numbered, used in medical/science
      TURABIAN = "turabian"               # Turabian (Chicago variant for students)
      CUSTOM = "custom"                    # Project-specific style
  ```
